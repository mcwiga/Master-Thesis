\setcounter{chapter}{0}
\chapter{Preliminaries}\label{ch:prelim}
The goal of this chapter is to provide some of the background needed to begin
our classification of Lagrangian fibrations over compact surfaces. We will
start with some of the theory of affine manifolds as these will prove to be a
crucial structure that the base space of such a fibration inherits. These
belong to a wider class of geometries called locally homogeneous geometric
structures. We will develop the theory of affine manifolds in the broader
context of these geometries.

This will lead us to consider representations of the fundamental group of such
manifolds through the \emph{holonomy representation}. We will spend sometime to
introduce results from the representation theory of affine manifolds which we
will later use in our classification results. After this, we specialize to the
case of integral affine manifolds as these will turn out to be an invariant of
the base space of a Lagrangian fibration.

\section{Geometric Structures on Manifolds}
\subsection{Basic definition and properties}
One of the main objects of study throughout this thesis are affine manifolds.
These belong to a class of geometries called \emph{locally homogeneous
    geometric structures}. The idea behind these structures is that we can describe
our geometry through charts into some model space $X$ and that we can stitch
these charts together through elements of some Lie group $G$ which acts
transitively on the model space.

We will require that our group $G$ be a Lie group acting transitively on $X$.

\begin{definition}\label{def:locally-g}
    Let $U \subset X$ be an open set of our model space $X$. We say that a smooth map $f: U \to X$
    is \textit{locally}-G if for each component $U_i \subset U$ there is a $g_i \in G$ such that
    $f|_{U_i} = g_i|_{U_i}$.

    We also require the map $f$ to satisfy the following unique extension property.
    If $U \subset X$ and $f: U \to X$ is a locally-$G$ map then there is a unique
    $g \in G$ restricting to $f$.
\end{definition}

\begin{definition}
    A $(G,X)$-atlas is a pair $(\mathcal{U}, \Phi)$  consisting of an open covering for $M$
    and a collection of charts such that the transition maps are locally-$G$ on their respective overlaps.
\end{definition}

\begin{definition}
    A map $f: M \to N$ between two $(G,X)$-manifolds is a $(G,X)$-map if for each pair of charts
    $(\phi, \varphi)$ then $\phi \circ f \circ \varphi^{-1}$ is locally-G on the relevant intersection.
\end{definition}
\begin{eg}
    A \emph{hyperbolic surface} $\Sigma$ is a $(G, X)$-structure where $X$ is the hyperbolic plane $\mathbb{H}^2$ and
    $G = \mathrm{PSL}(2, \R)$.
\end{eg}
\begin{eg}
    A \emph{real projective structure} on $M$ is a $(G, X)$-structure where $X = \R P^n$ and $G = \mathrm{PGL}(n+1, \R) = \dfrac{ \mathrm{GL}(n+1, \R)}{\R^{*}}$.
\end{eg}
\begin{eg}
    A \emph{flat Riemannian structure} or \emph{Euclidean structure} on $M$ is a $(G,X)$-structure where $G = O(n) \rtimes \R^n$ and $X = \R^n$.
\end{eg}

If $M$ is a $(G,X)$-manifold and $\pi: \tilde{M} \to M$ is a regular covering,
it is always possible to pull back the $(G,X$-structure on $M$ to $\tilde{M}$.

\begin{lemma}\label{lemma:pullback-affine-structure}
    Let $M$ be equipped with a $(G,X)$-structure and suppose that $\pi: \tilde{M} \to M$ is a regular covering map. Then
    the $(G,X)$-structure on $M$ can be pulled back to $\tilde{M}$.
\end{lemma}
\begin{proof}
    Since $\pi$ is a covering map for each $x \in M$ there exists an open set $U_x$ of $x$ such that $\pi^{-1}|_{U_x}$ is a
    disjoint union $\bigsqcup_i V_i$ and $\pi|_{V_i}$ is a homeomorphism. If necessary, we shrink the open set $U_x$ so that we may assume it lies entirely inside a $(G,X)$-chart $(W, \varphi)$.
    Now, doing this for each $x \in M$, we define charts on $\tilde{M}$ by $(V_i, \varphi \circ \pi|_{V_i})$. Since $\pi$ is surjective and restricts to a homeomorphism on each $V_i$, it is clear that this defines
    an atlas on $\tilde{M}$. It remains to check that our transition maps lie in $G$.

    Suppose $V_i \cap V_j \neq \emptyset$. Then $\pi(V_i) \cap \pi(V_j) \neq
        \emptyset$ but we know that they lie inside $(G,X)$-charts $(W_i, \varphi_i)$
    and $(W_j, \varphi_j)$ respectively with $W_i$ possibly equal to $W_j$. If they
    are distinct charts then since $M$ has a $(G,X)$-structure we know that their
    transition maps lie in $G$. But the transition maps for $(V_i, \varphi_i)$,
    $(V_j, \varphi_j)$ on $\tilde{M}$ are
    \begin{align*}
        (\varphi_i \circ \pi|_{V_i}) \circ (\varphi_j \circ \pi|_{V_j})^{-1} & = (\varphi_i \circ \pi|_{V_i})\circ (\pi|_{V_j}^{-1} \circ\varphi_j^{-1}) \\
                                                                             & = \varphi_i \circ \varphi_j^{-1}
    \end{align*}
    which is the transition map between $W_i$ and $W_j$ and hence in $G$ as $M$ has a $(G,X)$-structure.
\end{proof}

\subsection{The developing map and holonomy representation}
For $(G,X)$-manifolds, there is an inherently nice way to "globalize" the
charts via a $(G,X)$-map called the developing map. The study of this map will
be of importance to us throughout this thesis.

\begin{prop}
    Let $M$ be a simply connected $(G,X)$-manifold. Then there exists a $(G,X)$-map
    $$
        \text{dev}: M \to X.
    $$
\end{prop}

\begin{proof}
    We pick a base point $x_0 \in M$ and a chart $(U_0, \varphi_0)$ around $x_0$. Now for any $x \in M$ we define the map $\dev$ as follows. We pick a curve $x(t)$ on $M$ satisfying $x(0)=x_0$ and $x(1)=x$. As the image of our curve is compact, we can cover it by finitely many coordinate charts $(U_i, \varphi_i)$ with $i\in \{0, \dots,n\}$ such that $x(t) \in U_i$ for $t\in(a_i, b_i)$ with
    $$a_0<0<a_1<b_0<a_2<b_1<a_3 \dots <a_n<b_{n-1}<1<b_n$$

    \begin{figure}[H]
        \centering
        \incfig[0.7]{DevMap}
        \caption{Developing along a curve}
    \end{figure}

    \begin{figure}[H]
        \centering
        \incfig[0.6]{DevPartTwo}
        \caption{We successively glue our charts together using the transition maps}
    \end{figure}

    Now we consider where these charts overlap. Since our manifold is equipped with
    a $(G,X)$-structure our transition maps differ by an element of $G$. Let $g_i =
        \varphi_{i-1}\circ\varphi^{-1}_i \in G$ be the transition map between $U_i$ and
    $U_{i-1}$. We then define $$ \text{dev}(x) = g_1g_2\dots g_n \varphi_n(x) .$$
    We must show that this is a well-defined map. We first show that this map does
    not change if we refine the cover. Suppose we insert a chart $(V, \phi)$
    between $U_i$ and $U_{i-1}$. Let $h_i, h_{i-1} \in G$ be the unique group
    elements such that
    \begin{center}
        $\varphi_{i-1} = h_{i-1}\circ\phi$ on $V \cap U_{i-1}$ and \\
        $\phi = h_i\varphi_{i}$ on $V \cap U_{i-1}$.
    \end{center}
    Then on $V \cap U_i \cap U_{i-1}$ we have that $\varphi_{i-1} = h_{i-1}\circ h_i \circ \varphi_{i}$. This gives us that $g_i = \varphi_{i-1}\circ \varphi^{-1}_i = h_{i-1} \circ h_i$ as we require that our transition maps satisfy a unique extension property.

    Now when we develop along our curve with respect to this new covering, we
    obtain that:
    \begin{align*}
        \text{dev}(x) & = g_1 \dots g_{i-1}h_{i-1}h_{i}g_{i+1}\dots g_n\varphi_n(x) \\
                      & = g_1 \dots g_{i-1}g_ig_{i+1}\dots g_n\varphi_n(x)
    \end{align*}

    This defines $\text{dev}$ along $x(t)$. We now need to show that it does not
    depend on our choice of curve. As $M$ is simply connected, all curves with the
    same start point and end point will be homotopic. By compactness we can split
    our homotopy into smaller homotopies such that we can break up our curve into
    regions
    \begin{center}
        $a_1 =0 < a_2 < \dots < a_n=1$
    \end{center}
    such that during each small homotopy the segment $x((a_i, a_{i+1}))$ lies entirely inside a coordinate patch. It then follows that $\dev$ is independent of our choice of paths completing the proof.

\end{proof}

This shows that we have a well defined developing map on any simply connected
$(G,X)$-manifold. However, this map depended on the choice of base point and
initial chart. We immediately get the following corollary from this fact.

\begin{corollary}
    Let $M$ be a $(G,X)$-manifold and let $\tilde{M}$ be its universal cover. Then
    there exists a pair $(\text{dev}, \text{hol})$ satisfying
    \begin{center}
        $\text{dev}: \tilde{M} \to X$\\
        $\text{hol}: \pi_1(M) \to G$
    \end{center}
    where $\text{hol}(\gamma)$ is the element $g_1\dots g_n$ of $G$ obtained from
    developing around a loop $\gamma \in M$. The map $\dev$ is a local diffeomorphism and the map $\hol$ is a group homomorphism.
\end{corollary}

The development pair satisfy the following equivariance condition. Let $\gamma
    \in \pi_1(M)$ act by deck transformations on $\tilde{M}$. Then we have the
following.
\begin{lemma}
    Let $\gamma \in \pi_1(M)$. Then

    \begin{center}
        $\text{dev}(\gamma \cdot x) = \text{hol}(\gamma)\text{dev}(x)$
    \end{center}
    for $x \in \tilde{M}$.
    \label{lemma:dev-equivariance}
\end{lemma}

\begin{proof}
    Let $\gamma \in \pi_1(M)$ and let $\{(V_i, \phi_i)\}$ be a cover for $\gamma$. We pick a lift of $\gamma$ to the universal cover $\tilde{M}$ which we denote by
    $\tilde{\gamma}$. We can pull back our cover of $\gamma$ to a covering of the lift $\tilde{\gamma}$. Denote the transition maps on $V_i \cap V_{i+1}$ by $g_i$.
    We now wish to compute the action of $\gamma$ on a point $x \in \tilde{M}$.

    Letting $x_0\in \tilde{M}$ be the base point for our developing map, we pick a
    curve $\tilde{\alpha}$ joining $x_0$ to $x$. We consider the composition
    $\alpha \circ \gamma$. This has a unique lift to $\tilde{M}$ by choosing a lift
    of $\alpha$ such that $\alpha'(0) = \tilde{\gamma}(1)$. Now, concatenating
    these curves and evaluating the endpoint gives the action i.e. $\dev(\gamma
        \cdot x) = (\alpha' \cdot \tilde{\gamma})(1)$.

    \begin{figure}[H]
        \centering
        \incfig[0.4]{liftDevelopingMap}
        \caption{The action $\gamma \cdot x$.}
    \end{figure}

    Now, we note that starting at $x_0 \in \tilde{M}$ and first developing along
    $\tilde{\gamma}$ we pick up the group element $\hol(\gamma)$ by definition. We
    claim that starting at $\tilde{\gamma}(1)$ and developing along $\alpha'$ we
    pick up exactly $\dev(x)$. This is true because the lifts $\alpha'$ and
    $\tilde{\alpha}$ both project onto $\alpha$ by the covering map and since the
    $(G,X)-$structure on $\tilde{M}$ is defined by pulling back the one on $M$, it
    follows that developing along $\alpha'$ yields $\dev(x)$.
\end{proof}

The role of the development pair is to "globalize" the coordinate chart of $M$.
Indeed, the following proposition makes this statement clearer.

\begin{prop}
    Let $M$ be a $(G,X)$-manifold with development pair $(\dev, \text{hol})$. Denote
    by $\Gamma$ the image of $\pi_1(M)$ under the holonomy map. Then there
    exists a $(G,X)$-atlas such that the transition maps lie in $\Gamma$. \label{prop:dev-charts}
\end{prop}

\begin{proof}

    We pick $m\in M$ and fix a lift $\tilde{m}\in \tilde{M}$ of $m$. Let $p:
        \tilde{M} \to M$ be the universal cover. Then, as $\dev$ is a local
    diffeomorphism, there is an open set $\tilde{U}$ of $\tilde{m}$ such that
    $\dev$ is a diffeomorphism onto some open subset of $X$. By shrinking
    $\tilde{U}$ if necessary, we may assume that $V =p(\tilde{U})$ is evenly
    covered i.e. $p^{-1}(V) = \bigsqcup_{i \in I}V_{i}$ with $p$ a diffeomorphism
    on each $V_{i}$.

    We define coordinate charts on $M$ by $(V_{i}, \psi_{i} = \text{dev}\circ
        p^{-1}_{i})$ at a point $m \in M$ where $p_{i}: V_{i} \to V$ is a
    diffeomorphism.

    Now suppose that $V_i \cap V_j \neq \emptyset$ and let $\tilde{U}_i =
        p_i^{-1}(V_i)$ and $\tilde{U}_j = p_j^{-1}(V_j)$. Then the preimage of their
    intersection will differ by a deck transformation $\gamma \in \pi_1(M)$.
    Therefore, on the intersection $V_i \cap V_j$
    \begin{align*}
        \text{dev}\circ p_i^{-1}    & = \text{dev}\circ \gamma \circ p_j^{-1} \\
        \implies p^{-1}_i \circ p_j & = \gamma
    \end{align*}

    Now our by definition of our transition maps
    \begin{align*}
        \psi_i \circ \psi_j^{-1} & = \text{dev}\circ p_{i-1} \circ p_j \circ dev^{-1} \\
                                 & = \text{dev}\circ \gamma \circ \dev^{-1}           \\
                                 & = \text{hol}(\gamma)\dev\circ \dev^{-1}            \\
                                 & = \text{hol}(\gamma) \in \Gamma
    \end{align*}

    and so we have given our coordinate charts on $M$ in terms of the development
    pair.
\end{proof}

This shows us that our $(G,X)$-structure is determined by the development pair.
Conversely, the following lemma shows us that once we have fixed our
$(G,X)$-structure, the development pair is determined up to certain
factors/conjugations.

\begin{lemma}
    Let $M$ be a $(G,X)$-manifold with development pair $(\text{dev}, \text{hol})$. Suppose that $(\text{dev}', \text{hol}')$ is
    another development pair for $M$. Then $\text{dev}' = g\circ \text{dev}$ and $\text{hol}' = g\circ \text{hol}\circ g^{-1}$
    for some $g \in G$. \label{lemma:equivalent-dev-pair}
\end{lemma}
\begin{proof}
    Our development pair $(\dev, \hol)$ was determined once we picked a base point $x_0 \in \tilde{M}$ and a chart $(U, \phi)$ of $x_0$. Suppose that
    we pick a different base point $y_0 \in \tilde{M}$ and initial chart $(V, \varphi)$. This determines a new development pair $(\dev, \hol')$ and we wish to determine the relationship between the two pairs.

    We pick a curve $\gamma$ joining $y_0 \mapsto x_0$. For a point $x \in
        \tilde{M}$, we can now define $\dev'(x)$ by picking a cover $\{V_i\}_{i \in I}$
    of the curve $\gamma$, developing along $\gamma$ with respect to this cover,
    and then developing along any curve connecting $x_0$ to $x$. However, by
    definition of the developing map, if we develop along the curve $\gamma$ we
    pick up some group element $g \in G$ by composing the transition maps of the
    covering. Then, when we develop from $x_0$ to $x$, this is by definition
    exactly the same as the value of our initial developing map $\dev(x)$. Hence we
    have that $\dev'(x) = g \dev(x)$ for some $g \in G$.

    Now to show that $\hol' = g\circ \text{hol}\circ g^{-1}$ we consider
    \begin{align*}
        \dev'(\gamma \cdot \alpha) & = g \circ \text{dev}(\gamma \cdot \alpha)                            \\
                                   & = g\circ \text{hol}(\gamma) \circ \text{dev}(\alpha)                 \\
                                   & = (g\circ \text{hol}(\gamma)\circ g^{-1})(g\circ \text{dev}(\alpha)) \\
                                   & = (g \circ \text{hol}(\gamma)\circ g^{-1})\text{dev}'(\alpha)
    \end{align*}
    and as the development pair satisfied the equivariance property it follows that
    $\hol' = g\circ \text{hol}\circ g^{-1}$.
\end{proof}

This shows that once we have fixed a development pair, any other development
pair is related by $(\text{dev}', \text{hol}') = (g\circ \text{dev}, g\circ
    \text{hol}\circ g^{-1})$. We should therefore note that if we want to classify
$(G,X)$-structures on a manifold $M$ by classifying the possible development
pairs, it is important to consider them up to this relation.

\section{Affine Manifolds}
Of primary interest to us throughout this thesis are \emph{affine structures}
on a manifold $M$. These are $(G,X)$-structures whereby $G$ is the group of
affine transformations of $\R^n$ and the model space $X$ is simply $\R^n$.
There are many interesting properties and open questions surrounding this class
of geometric structures which we shall see throughout this thesis. For now we
give some of the basic definitions and properties regarding these objects.

\begin{definition}
    An affine transformation of $\mathbb{R}^n$ is a map
    \begin{center}
        $A: \mathbb{R}^n \to \mathbb{R}^n$
    \end{center}
    such that $A(x) = Gx + b$ where $G \in GL(\mathbb{R}^n)$ and $b,x \in\mathbb{R}^n$.
\end{definition}
\begin{definition}[Affine Group]
    The group $$\text{Aff}(\mathbb{R}^n) =
        \text{GL}(\mathbb{R}^n )\rtimes \mathbb{R}^n$$ is the affine group of $\R^n$ and consists of all the affine transformations of
    $\mathbb{R}^n$.
\end{definition}

\subsection{Affine manifolds and connections}

In $\R^n$ we can differentiate vector fields $X, Y \in \mathfrak{X}(M)$ by
taking the directional derivative $\nabla_X Y$. One can extend the
differentiation of vector fields in $\R^n$ to arbitrary manifolds through the
notion of a \emph{connection}. One can define a connection for any vector
bundle but we will focus only on the case our vector bundle is the tangent
bundle of a smooth manifold.

\begin{definition}
    Let $M$ be a smooth manifold and denote by $TM$ its tangent bundle. Then a connection on $TM$ is a map
    $$
        \nabla : \mathfrak{X}(M) \times \mathfrak{X}(M) \to \mathfrak{X}(M)
    $$
    which satisfies the following properties.
    \begin{enumerate}
        \renewcommand{\labelenumi}{\roman{enumi})}

        \item $\nabla_X Y$ is linear over $C^\infty(M)$ in $X$: for
              $f_1,f_2 \in C^\infty(M)$ and $X_1,X_2 \in \mathfrak{X}(M)$,
              \[
                  \nabla_{f_1X_1+f_2X_2}Y
                  =
                  f_1\nabla_{X_1}Y+f_2\nabla_{X_2}Y.
              \]

        \item $\nabla_X Y$ is linear over $\mathbb{R}$ in $Y$: for
              $a_1,a_2 \in \mathbb{R}$ and $Y_1,Y_2 \in \mathfrak{X}(M)$,
              \[
                  \nabla_X(a_1Y_1+a_2Y_2)
                  =
                  a_1\nabla_XY_1+a_2\nabla_XY_2.
              \]

        \item $\nabla$ satisfies the following product rule: for
              $f \in C^\infty(M)$,
              \[
                  \nabla_X(fY)
                  =
                  f\,\nabla_XY+X(f)Y.
              \]

    \end{enumerate}
\end{definition}
\begin{remark}
    The term connection comes from the fact that a connection provides a
    way to identify fibres in a vector bundle by transporting a basis along
    a curve in a way that is covariantly constant with respect to the connection.
\end{remark}
\begin{definition}[Christoffel Symbols]
    Let $E_1, \dots , E_n$ be a frame on $M$ and let $\nabla$ be a connection in $TM$. Then we have that
    $$
        \nabla_{E_i}E_j = \sum_k \Gamma_{ij} E_k
    $$
    for some smooth functions $\Gamma_{ij}^k$. These functions are called the \emph{Christoffel symbols} of $\nabla$.
\end{definition}
\begin{definition}
    A vector field $X \in \mathfrak{X}(M)$ will be called \emph{parallel} with respect to $\nabla$ if
    $$\nabla_Y X =0$$
    for all $Y \in \mathfrak{X}(M)$. We will write this condition as $\nabla X = 0$.
\end{definition}

Fixing an affine structure on $M$ will turn out to be equivalent to equipping
$M$ with a certain type of connection. In order to show this we need to
introduce the notion of curvature and torsion of a connection.

\begin{definition}[Torsion tensor]
    The \emph{torsion tensor} of a connection $\nabla$ on $TM$ is defined to be
    $$
        \tau(X,Y) = \nabla_X Y - \nabla_Y X - [X,Y]
    $$
    where $X,Y \in \mathfrak{X}(M)$.
\end{definition}

Next, we introduce the notion of curvature which measures the failure of
differentiating with respect to the connection $\nabla$ to commute.

\begin{definition}[Curvature Tensor]
    The \emph{curvature tensor} of a connection $\nabla$ on $TM$ is defined as
    $$
        R(X,Y)(Z) = \nabla_X \nabla_Y Z - \nabla_Y \nabla_X Z - \nabla_{[X,Y]}Z
    $$
    where $X,Y,Z \in \mathfrak{X}(M)$.
\end{definition}
\begin{definition}
    A connection will be called \emph{flat} if $R(X,Y)Z = 0$ and \emph{torsion free} if $\tau(X,Y)=0$ for all $X,Y, Z \in \mathfrak{X}(M)$.
\end{definition}

We now illustrate the link between affine structures on $M$ and connections on
$TM$.

\begin{lemma}
    An affine structure on a smooth manifold $M$ is equivalent to a flat, torsion free connection on $TM$.
\end{lemma}
\begin{proof}
    Let $M$ be an affine manifold and $\tilde{M}$ be its universal cover. Then since $\dev$ is a local diffeomorphism, we can pull back the standard flat, torsion free
    connection $\nabla$ on $\R^n$ to obtain a flat, torsion free connection on $\tilde{M}$ denoted by $\dev^* \nabla$. We now show that this connection descends to $M$.

    We show that $\nabla$ is invariant under deck transformations. We have that
    \begin{align*}
        \gamma^* \dev^* \nabla & = (\dev \circ \gamma)^* \nabla       \\
                               & = (\hol(\gamma) \circ \dev)^* \nabla \\
                               & = \dev^* \hol(\gamma)^* \nabla
    \end{align*}
    where $\gamma \in \pi_1(M)$. But affine maps preserve the euclidean connection as they preserve the notion of zero acceleration and so
    $$
        \dev^*\hol(\gamma)^* \nabla = \dev^*\nabla
    $$
    and hence deck transformations preserve the connection and it descends to $M$.

    Conversely, suppose that $M$ is equipped with a flat, torsion free connection
    $\nabla$. Let $e_1, \dots , e_n$ be a basis for $T_pM$ for some $p \in M$. Then
    since the connection $\nabla$ is flat, there is an open neighborhood $U \subset
        M$ of $p$ such that parallel transport by $\nabla$ is path-independent. Hence,
    we parallel transport $e_i$ using $\nabla$ to get a frame $E_1, \dots , E_n$ on
    $U$. Now, since $\nabla$ is torsion-free and the vector fields $E_i$ are
    parallel, we have that
    \begin{align*}
        [E_i, E_j] = 0.
    \end{align*}
    Now, Theorem  $9.46$ of \cite{lee2003introduction} implies that this frame gives rise to coordinates $x^1, \dots , x^n$ on $U$ such that
    $\frac{\partial}{\partial x^i} = E_i$. Now, suppose that $y^1, \dots , y^n$ are coordinates defined in the same way on $V$. The transformation
    law for the Christoffel symbols yields
    \begin{align*}
        \tilde{\Gamma}_{ij}^k = \dfrac{\partial y^k}{\partial x^m}\Gamma_{ij}^m + \dfrac{\partial^2 y^k}{\partial x^i \partial x^j}
    \end{align*}
    where we sum over repeated indices. But since the $\frac{\partial}{\partial x^i}$ and $\frac{\partial}{\partial y^i}$ are parallel, i.e. $\nabla \frac{\partial}{\partial x^i}= \nabla \frac{\partial}{\partial y^i} = 0$, we have that $\Gamma_{ij}^m =  \tilde{\Gamma}_{ij}^k =0$ and hence
    $$
        \dfrac{\partial^2 y^k}{\partial x^i \partial x^j} = 0.
    $$
    Integrating this gives that $y^k = A_i^k x^i + b^k$ and hence our transition maps are affine maps.
\end{proof}
\begin{remark}
    In local affine coordinates, this connection is just the usual covariant derivative in $\R^n$.
\end{remark}

\subsection{Completeness of affine structures}
Associated to a connection, one has the notion of \emph{geodesics}. Given a
curve $\gamma: (0,1) \to M$ one can associate its velocity vector field
$\dot{\gamma}$ which is a vector field along $\gamma$. To avoid some
technicalities, we will assume that this velocity vector field has an extension
to a vector field on all of $M$.

\begin{definition}
    Let $M$ be a smooth manifold and let $\gamma: (0,1) \to M$ be a curve on $M$. Then $\gamma$ is
    a \emph{geodesic} if $\nabla_{\dot{\gamma}} \dot{\gamma} =0$.
\end{definition}
\begin{definition}
    A connection $\nabla$ is said to be \emph{complete} if its geodesics are defined for all time $t$.
\end{definition}
\begin{remark}
    The Euclidean plane $\R^2$ with the usual connection is complete as its geodesics are straight lines and can be continued indefinitely.
    However, if we remove the origin $0 \in \R^2$, then $\R^2 \setminus 0$ is no longer complete with its usual connection as a straight line cannot be continued through the origin.
\end{remark}

For affine manifolds, there is a relationship between this notion of
completeness and a different notion of completeness coming from the theory of
$(G,X)$-manifolds.
\begin{definition}\label{def:dev-complete}
    A $(G,X)$-manifold $M$ is said to be complete if its developing map is a diffeomorphism
    from $M$ to $X$.
\end{definition}

\begin{remark}
    In the case that $M$ is a Riemannian manifold, this notion of completeness is
    equivalent to completeness of the Levi-Civita connection of $M$.
\end{remark}

\begin{prop}\label{prop:complete-if-quotient}
    Let $M$ be a $(G,X)$-manifold with $X$ simply connected. Then $M$ is complete in the sense of Definition \ref{def:dev-complete}
    if and only if $M = X/ \Gamma$ with $\Gamma \subset G$.
\end{prop}
\begin{proof}
    Suppose that $M = X/ \Gamma$. Then covering space theory tells us that $\pi: X \to X / \Gamma$  is its universal cover and is $\Gamma$ its group of deck transformations.
    Letting $\dev = \text{id} : X \to X$ the equivariance property is now satisfied trivially by $\Gamma$.

    Conversely, suppose that $\dev : \tilde{M} \to X$ is a diffeomorphism. Then, by
    equivariance of the holonomy representation $\hol: \pi_1(M) \to G$, we have $$
        \dev(\gamma \cdot x) = \hol(\gamma) \dev(x) $$ where $\gamma \in \pi_1(M)$ acts
    by deck transformations. Hence, deck transformations correspond to the action
    by the holonomy group under this identification. Since deck transformations act
    freely and properly on the universal cover of $M$, it follows as $\dev$ is a
    diffeomorphism that $\hol (\pi_1(M)) \subset G$ acts freely and properly on
    $X$. By covering space theory $\tilde{M}/ \pi_1(M) = M$ and since by
    completeness and equivariance of $\dev$ we have $\tilde{M}/ \pi_1(M) \cong X /
        \hol (\pi_1(M))$ the result follows.
\end{proof}

For affine manifolds we have the following equivalence of completeness in the
sense of Definition \ref{def:dev-complete} and geodesic completeness of the
associated flat, torsion-free connection.

\begin{theorem}
    Let $M$ be an affine manifold and let $\nabla$ be its associated flat, torsion-free connection. Then $\nabla$ is
    geodesically complete if and only if $\dev : \tilde{M} \to \R^n$ is a diffeomorphism.
\end{theorem}
\begin{proof}
    If $\dev$ is a diffeomorphism and $D$ is the standard Euclidean connection on $R^n$ then $\dev^* D$ is complete since there is a one-to-one correspondence between
    geodesics on $\tilde{M}$ and on $\R^n$. Now suppose that the descension $\nabla$ of $\dev^* D$ to $M$ is incomplete then there would be a geodesic on $M$ that is incomplete. But this would
    lift to an incomplete geodesic on $\tilde{M}$ but $(\tilde{M}, \dev^* D)$ is complete.

    Conversely, suppose that $\nabla$ is complete. Since if $M$ is complete so is
    its universal cover, we may assume $M$ is simply connected. Define the
    exponential map by $$ \text{exp}_p: U \subset T_pM \to M $$ $$ \text{exp}_p(tv)
        = \gamma(t) $$ where $\gamma$ is a geodesic with initial conditions $(p, v) \in
        M \times T_pM$. It is well known, for instance see \cite{goldman2022geometric},
    that $M$ is complete if and only if $\text{exp}_p$ is defined on all of $T_pM$
    for all $p \in M$. The following diagram can be seen to commute;
    \[
        \begin{tikzcd}[column sep=4em, row sep=3em]
            T_pM
            \arrow[r, "{D(\mathrm{dev})_p}"]
            \arrow[d, "\mathrm{exp}_p"']
            &
            T_{\mathrm{dev}(p)}\R^n
            \arrow[d, "\mathrm{exp}_{\dev(p)}"]
            \\
            M
            \arrow[r, "\mathrm{dev}"']
            &
            \R^n
        \end{tikzcd}.
    \]
    Indeed, let $\gamma: (0,1) \to M$ be a geodesic with initial conditions $(p,v)
        \in M \times T_pM$. Then
    \begin{align*}
        \dev \circ \text{exp}_p(tv) = \dev (\gamma(t)).
    \end{align*}
    On the other hand,
    \begin{align}
        \text{exp}_{\dev (p)}D(\dev)_p(tv) = \left.\text{exp}_{\dev (p)} \dfrac{d}{dt}\right\vert_{t=0}\dev (\gamma(t))
    \end{align}
    But $\dev(\gamma(t))$ is a geodesic since $\nabla = \dev^* D$ and hence
    $$ \left.\text{exp}_{\dev (p)} \dfrac{d}{dt}\right\vert_{t=0}\dev (\gamma(t)) = \dev (\gamma(t)).$$
    Since $D(\mathrm{dev})_p$ and $\text{exp}$ are bijections it follows that
    $\dev$ is also a bijection and hence a diffeomorphism.
\end{proof}

\subsection{Parallel Volume}

We finish the discussion of connections on affine manifolds with the notion of
a \emph{parallel volume form}. Given a connection on $TM$ one can also obtain a
connection on the $k$-forms on $M$ through the following Koszul-like formula
$$(\nabla_X \omega)(X_1,\dots,X_k) = X\!\bigl(\omega(X_1,\dots,X_k)\bigr) -
    \sum_{i=1}^k \omega(X_1,\dots,\nabla_X X_i,\dots,X_k)$$ where $\omega \in
    \Omega^k(M)$ and $X, X_i \in \mathfrak{X}(M)$. It is conjectured in
\cite{markus1963cosmological} that if an affine manifold possesses a
\emph{parallel volume form} then it is complete. This is now referred to as the
\emph{Markus conjecture}. By parallel volume form we mean a volume form $\omega
    \in \Omega^n(M)$ such that $\nabla \omega \equiv 0$.

Corollary \ref{lemma:volume} implies that $M$ has parallel volume if and only
if every element of its linear holonomy group is volume preserving. Another
characterisation which arises from this observation is the following. If $M$ is
orientable, from $$ \pi_1(M) \xrightarrow{\text{Lin} \circ \hol}
    \mathrm{GL}(\R^n) \xrightarrow{\log \circ|\det| } \R $$ we see that there is an
additive homomorphism $\nu_M \in \Hom(\pi_1(M), \R) \cong H^1(M, \R)$ called
the \emph{volume obstruction} and $M$ possesses a parallel volume form if and
only if $\nu_M = 0$.

This question is still open in full generality, however it has been solved in
some specific cases. One of the main tasks of this thesis is to show that the
Markus conjecture is true in the case of nilpotent holonomy. This will prove a
powerful tool in classifying affine structures as completeness implies that the
affine manifold is a quotient of its model space.
\subsection{Affine Structures on the Torus}
The standard affine structure on the torus is obtained from viewing the torus
as $\mathbb{R}^n/\mathbb{Z}^n$ where $\mathbb{Z}^n$ acts by translations. This
structure is easily seen to be a complete integral affine manifold. Indeed, the
action is an action by the integral affine group and is both free and proper.
Covering space theory tells us that $\pi : \mathbb{R}^n \to
    \mathbb{R}^n/\mathbb{Z}^n$ is a covering space with the group of deck
transformations via our $\mathbb{Z}^n$-action. In \cite{MarkusAuslander} it is
shown that all complete affine structures are obtained in this manner. However,
even for the 2-torus, not all affine structures are complete. We now give an
example of an incomplete structure on $\mathbb{T}^2$.

\begin{eg}\label{eg:hopf-manifold}
    Let $\lambda >1$ be a real number and consider the action of $\Z$ generated by multiplication by $\lambda$ on $\mathbb{R}^2- \{0\}$. This action is a free and proper action by the affine group and so the quotient
    $M = \mathbb{R}^2- \{0\} / \langle\lambda\rangle$ is an affine manifold.

    \begin{figure}[H]
        \centering
        \incfig[0.4]{hopftorus}
        \caption{Visualisation of the Hopf manifold. The points $1$ and $\lambda$ are identified.}
    \end{figure}

    The map $$\rho : \mathbb{R}^2 \to \mathbb{R}^2- \{0\}$$ $$(x,y) \mapsto e^x
        (\cos y, \sin y)$$ is a covering map. Basing our fundamental group at $[(1,0)]
        \in M$, our group is generated by the loops $\gamma_1$ and $\gamma_2$ where
    $\gamma_1$ is the projection of the line segment joining the point $(1,0)
        \mapsto (\lambda, 0)$ and $\gamma_2$ is the loop going around the unit circle
    once. These lift to curves $\bar{\gamma}_1$ and $\bar{\gamma}_2$ in the
    universal cover satisfying $$\bar{\gamma}_1(0) = (0,\ln(1)) = (0,0), \quad
        \bar{\gamma}_1(1) = (0,\ln(\lambda)) ,$$ $$\bar{\gamma}_2(0) = (0, 0) , \quad
        \bar{\gamma}_2(1) = (0, 2\pi),$$ and from this we see that $\gamma_1$ and
    $\gamma_2$ act on $(x,y)$ by $$\gamma_1 \cdot (x,y) = (\ln(\gamma) + x, y)$$
    $$\gamma_2(x,y) = (x, y + 2\pi).$$

    Now, we define our developing pair by $$\dev(x,y) = \rho(x,y),$$
    $$\hol(\gamma_1)\cdot (x,y) = (\lambda x, \lambda y),$$ $$\hol(\gamma_2) =
        \text{id}_{\mathbb{R}^2}.$$

    We check that these satisfy the equivariant property;
    \begin{align*}
        \dev(\gamma_1\cdot(x,y)) & = \dev((\ln(\lambda) + x,y))          \\
                                 & = e^{\ln(\lambda)}e^x(\cos y, \sin y) \\
                                 & = \lambda \dev((x,y))                 \\
                                 & = \hol(\gamma_1) \dev((x,y)),
    \end{align*}
    \begin{align*}
        \dev(\gamma_2\cdot(x,y)) & = \dev((x,y + 2\pi))                  \\
                                 & = e^x(\cos (y+ 2\pi), \sin (y+ 2\pi)) \\
                                 & = e^x(\cos y, \sin y)                 \\
                                 & = \hol(\gamma_2) \dev((x,y)).
    \end{align*}
    It is now clear that this structure is incomplete as $0 \notin \dev(\mathbb{R}^2)$. Indeed, it is typical that the fixed-point of a radiant structure is not in the developing image as shown in Theorem 6.5.2 \cite{goldman2022geometric}.
\end{eg}

\section{Group (Co)homology}
A theme throughout this thesis will be that the holonomy representation of an
affine manifold will encode many important properties of the affine structure.
In order to illustrate this, we will need to develop some of the theory of
affine representations. These results will require some machinery from the
theory of group (co)homology. The aim of this section is to provide a brief
overview of these theories although it will be far from a complete discussion.
We will also assume a basic familiarity with modules over a ring. We refer the
reader to \cite{Weibel_1994} and \cite{brown2012cohomology} for a more in-depth
review.

We start with the notion of the group ring.

\begin{definition}[Group Ring]
    Let $G$ be any group. The group ring $\Z[G]$ is the free abelian group generated by the elements of $G$. The multiplication
    is defined by
    $$
        \left(\sum_{h \in G}m_h h\right)\left(\sum_{k \in G}n_k k\right) = \sum_{g \in G}\left(\sum_{hk = g}m_hn_k\right)g.
    $$
\end{definition}

\begin{definition}
    A $\Z[G]$-module $M$ is called trivial if $g \cdot m = m$ for all $m \in M$ and $g \in G$.
\end{definition}
\begin{remark}
    We can always view an abelian group $A$ as a trivial $\Z[G]$-module. We will often use this to view $\Z$ as a $\Z[G]$-module.
\end{remark}

\begin{definition}
    A morphism of $\Z[G]$-modules $M$ and $N$ is a homomorphism of the underlying abelian groups
    $$
        \varphi : M \to N
    $$
    such that $\varphi(g \cdot m) = g \cdot \varphi(m)$ for all $g \in G$ and $m \in M$.
\end{definition}
\begin{eg}[Augmentation map]
    Let $M$ be a $\Z[G]$-module. Then there is a map
    $$
        \epsilon : M \to \Z
    $$
    $$
        \sum_{g \in G}m_g g\mapsto \sum_{g \in G} m_g
    $$
    where $\Z$ is viewed as a trivial $\Z[G]$-module.
\end{eg}
\begin{definition}
    The kernel of this map $\ker \epsilon := IG$ is called the \emph{augmentation ideal}.
\end{definition}

\begin{definition}
    A \emph{chain complex} of $\Z [G]$-modules is a series of $\Z[G]$-modules and $\Z[G]$ morphisms
    $$
        \dots \to C_{n+1} \xrightarrow{d_{n+1}} \to C_n \xrightarrow{d_n} C_{n-1} \to \dots
    $$
    such that $d_n \circ d_{n+1} = 0$. This implies that $\text{Im }d_{n+1} \subseteq \ker d_{n}$. We will denote this complex by
    $C_{\bullet}$.
\end{definition}
\begin{definition}
    The \emph{homology} of a chain complex $C_{\bullet}$ is defined as
    $$
        H_n(C_{\bullet}) = \dfrac{\ker d_n}{\text{Im }d_{n+1}}.
    $$
\end{definition}

If $M$ is a $\Z[G]$-module, then we can apply $\Hom_{\Z[G]}(\Z, M)$ and since a
module morphism from the trivial $\Z[G]$-module $\Z$ to $M$ is determined by
the image of $1 \in \Z$, we have that
\begin{align*}
    \Hom_{\Z[G]}(\Z, M) \cong \{m \in M \mid gm = m, \enskip \forall g \in G\}.
\end{align*}
The functor $\Hom_{\Z[G]}(\Z, -)$ maps between the category of $\Z[G]$-modules and abelian groups. This is called the \emph{invariants} functor. This is always left exact but is not right exact
in general. The idea of group cohomology is to measure the failure of this functor to be exact. The invariants functor is an example of a contravariant \emph{additive functor} which means that it preserves
direct sums. For these types of functors, assuming the category you are working in has enough injectives/projectives, you can always associate a cohomology/homology theory which measures the failure of the functor to be exact. For those already familiar with some homological algebra,
the group cohomology of a $\Z[G]$-module $M$ is the right derived functor of the invariants functor $\Hom_{\Z[G]}(\Z, -)$. However, we will give a more concrete definition of this in what follows.

\begin{definition}
    Let $M$ be a $\Z[G]$-module. A \emph{free resolution} of $M$ by $\Z[G]$-modules is an exact sequence
    $$
        \dots \to F_1 \to F_0 \to M \to 0
    $$
    where each of the $F_i$ is a free $\Z[G]$-module.
\end{definition}
\begin{remark}
    Recall that a free $R$-module $M$ is an $R$-module such that $M \cong \bigoplus_{i \in I} R$.
\end{remark}
The group cohomology of a $\Z[G]$-module can be calculated by taking a
resolution of the trivial $\Z[G]$-module $\Z$ by free $\Z[G]$-modules,
applying $\Hom_{\Z[G]}(-, M)$ to this resolution and
calculating the homology of the resulting chain complex.
\begin{definition}[Group Cohomology]
    Let $M$ be a $\Z[G]$-module. Then we can define the group cohomology $H^n(G,M)$ as follows. Take a free resolution
    $$
        \dots \to F_2 \to F_1 \to F_0 \to \Z \to 0
    $$
    of $\Z$ by $\Z[G]$-modules. Delete the trailing $\Z$ from the resolution to form the \emph{deleted resolution}
    $$
        \dots \to F_2 \to F_1 \to F_0 \to 0.
    $$

    We apply $\Hom_{\Z[G]}(-, M)$ to this resolution to obtain a new complex $$
        \dots \xleftarrow{d^{n+1}} \Hom_{\Z[G]}(F_n, M) \xleftarrow{d^n} \dots
        \xleftarrow{d^2} \Hom_{\Z[G]}(F_1, M) \xleftarrow{d^1} \Hom_{\Z[G]}(F_0, M)
        \xleftarrow{} 0. $$

    The cohomology group $H^i(G, M)$ is now the homology of this complex.
    Concretely, $$ H^i(G, M) = \dfrac{\ker d^{n+1}}{\Im d^n} .$$
\end{definition}
\begin{definition}[Group Homology]
    Let $M$ be a $\Z[G]$-module. The \emph{group homology} $H_i(G,M)$ is defined in the same was as the group cohomology except instead of
    applying $\Hom_{\Z[G]}(\Z, -)$ to the deleted resolution, we apply $- \otimes_{\Z[G]} M$ and take the homology of the resulting complex.
\end{definition}

We will make use of the following proposition which gives explicit formula for
the groups $H^0(G,M)$ and $H_0(G,M)$.

\begin{prop}\label{prop:group-zero-homol-cohom}
    Let $M$ be a $G$-module. Then
    \begin{enumerate}
        \item $H_0(G,M) = \dfrac{M}{(IG)M}$ where $IG$ is the augmentation ideal. This is the largest quotient of $M$ on which $G$ acts trivially since $(IG)M$ is generated by elements of the form $gm-m$.
        \item $H^0(G,M) := M^G = \{m\in M \mid gm = m, \enskip \forall g \in G\}$.
    \end{enumerate}
\end{prop}
\begin{proof}
    We first show that $H_0(G,M) = \dfrac{M}{(IG)M}$. We take the deleted free resolution and apply $- \otimes_{\Z[G]} M$ and obtain
    $$
        \dots \to F_2 \otimes M \to F_1 \otimes M \xrightarrow{d} F_0 \otimes M \to 0.
    $$
    We will use that $H_0(G, M)$ is the left derived functor of $\Z \otimes_{\Z[G]} -$, see for example \cite{brown2012cohomology}, so the degree zero cohomology
    is simply the functor itself i.e.
    $$
        H_0(G, M) = \Z \otimes_{\Z[G]} M.
    $$
    Now we have the short exact sequence
    $$
        0 \to IG \to \Z[G] \xrightarrow{\epsilon} \Z \to 0
    $$
    and so $\Z \cong \dfrac{\Z[G]}{IG}$. Now
    \begin{align*}
        H_0(G, M) & = \Z \otimes_{\Z[G]} M                \\
                  & = \dfrac{\Z[G]}{IG} \otimes_{\Z[G]} M \\
                  & = \dfrac{M}{(IG)M}.
    \end{align*}

    To show $H^0(M, G) = M^G$ we note that $H^0(M, G) = \Hom_{\Z[G]} (\Z, M)$. Now
    any $f \in \Hom_{\Z[G]} (\Z, M)$ is determined by $f(1)$. We define $$ \varphi
        : \Hom_{\Z[G]} (\Z, M) \to M^G $$ $$ f \mapsto f(1).$$

    Since $Z$ is viewed as a trivial $\Z[G]$-module
    \begin{align*}
        f(1) & =f(g\cdot 1) \\
             & = g f(1)
    \end{align*}
    and hence this does map into $M^G$. Injectivity and surjectivity is clear and hence this is an isomorphism.
\end{proof}
\section{Representation Theory and Affine Manifolds}

\subsection{Affine representations}
We now turn our attention to affine manifolds before specializing to the
integral affine case. An affine manifold is a $(G,X)$-manifold where $X =
    \mathbb{R}^n$ and $G = \text{Aff}(\mathbb{R}^n)$. An important tool in
classifying affine structures is the representation theory of groups. Often, by
studying properties of the holonomy representation of an affine manifold $M$,
we are able to obtain information about the affine structure of $M$. Of
fundamental importance is the case where the holonomy group is nilpotent which
we shall see in Chapter \ref{ch:nilpotent-hol}. We will use these tools to
determine conditions for the affine structure to be complete. We begin with the
notion of an \emph{affine representation}.

\begin{definition}
    An affine representation of a group $G$ is a group homomorphism
    \begin{center}
        $\alpha: G \to \text{Aff}(\mathbb{R}^n)$.
    \end{center}
\end{definition}

\begin{eg}
    Let $M$ be a $(G,X)$-manifold with $G = \text{Aff}(\mathbb{R}^n)$ and $X = \R^n$. Then the holonomy representation
    is an affine representation of $\pi_1(M)$.
\end{eg}
\begin{definition}
    Let $G$ be any group and let $A$ be an abelian group. We will call $A$ a $G$-module if there is a group homomorphism
    $$\rho: G \to \text{Aut}(A).$$
\end{definition}
\begin{remark}
    A $G$-module $A$ is a module in the usual sense over the group ring $\Z[G]$ which consists of formal integer sums of elements of $G$.
\end{remark}
\begin{remark}
    An affine representation $\alpha$ splits into a linear component and a
    translation component, i.e. $\alpha(g) = (\lambda(g), \mu(g))$ where $\lambda:
        G \to \text{GL}(\R^n)$ and $\mu : G \to \R^n$. The linear component makes
    $\mathbb{R}^n$ into a $G$-module. We will denote this $G$-module by $E$ when we want to emphasize that we are viewing $\R^n$ as a $G$-module.
\end{remark}
Analogous to irreducible representations in the usual sense, there is a notion of irreducibility for affine representations.
\begin{definition}
    An affine representation $\alpha: G \to \aff(E)$ is said to be \emph{irreducible} if $\alpha$ leaves a proper affine subspace
    $F \subset E$ invariant.
\end{definition}
\begin{remark}
    Note that unlike standard representations, for affine representations we do not allow an origin to be fixed in the notion of irreducibility as
    there is no concept of an origin for affine spaces.
\end{remark}

We now wish to illustrate that completeness of the affine structure has
implications on the group theoretic properties of the affine holonomy group. We
will use freely some of the properties of Eilenberg Mac-Lane spaces which can
be found in \cite{Hatcher} although we will recall the definition.
\begin{definition}\label{def:eilenberg-mac}
    Let $G$ be a group and let $n \geq 1$ be an integer. A path-connected topological space $X$ is called an \emph{Eilenberg--MacLane space} of type $K(G,n)$ if
    \[
        \pi_i(X) \cong
        \begin{cases}
            G & \text{if } i = n,    \\
            0 & \text{if } i \neq n.
        \end{cases}
    \]
    Equivalently, $X$ has a single nontrivial homotopy group, which occurs in
    degree $n$ and is isomorphic to $G$.
\end{definition}

\begin{theorem}\label{thm:complete-implies-irreducible}
    Let $M$ be a compact complete affine manifold. Then the affine holonomy representation is irreducible.
\end{theorem}
\begin{proof}
    Since $M$ is complete we have that $M = \R^n/\Gamma$ where $\Gamma$ is the affine holonomy group. Now, if $\Gamma$ is not
    irreducible, then there is an invariant affine subspace $V \subset \R^n$. Since $V$ is invariant under $\Gamma$ and $\Gamma$
    acts freely and properly on $\R^n$ it follows that $V/\Gamma$ is an affine manifold with universal cover $V$. Since both $\R^n$ and
    $V$ are contractible, it follows that $V/\Gamma$ and $M$ are Eilenberg Mac-Lane $K(\Gamma, 1)$ spaces.

    Let $i: V \to \R^n$ be the inclusion. The induced map $$ i_* : \pi_1(V/\Gamma)
        \to \pi_1(M) $$ is an isomorphism of fundamental groups and hence since both
    spaces are $K(\Gamma, 1)$ this is a homotopy equivalence by Whitehead's
    theorem. Since $\R^n / \Gamma$ and $V/\Gamma$ are both compact, it follows that
    $\dim V/ \Gamma = \dim \R^n/\Gamma$ and hence $V = \R^n$.
\end{proof}

We will need some more machinery from the theory of group cohomology. This
leads us to the following definitions.

\begin{definition}[Crossed Homomorphism]
    Let $G$ be a group and $E$ a $G$-module. A crossed homomorphism for $\lambda : G \to \text{Aut}(E)$ is a group homomorphism
    \begin{center}
        $\mu: G \to E$
    \end{center}
    that satisfies $\mu(gh) = \mu(g) + \lambda(g)\mu(h)$.
\end{definition}

\begin{prop}
    Let $\alpha = (\lambda, \mu)$ be an affine representation on $E$ of a group $G$. Then
    $\mu$ is a crossed homomorphism for $\lambda$.
\end{prop}
\begin{proof}
    Let $g,h \in G$ and $\alpha = (\mu, \lambda)$ be an affine representation. Then
    \begin{align*}
        \alpha(gh)       & = (\mu(gh), \lambda(gh))                            \\
                         & =\alpha(g)\cdot \alpha(h)                           \\
                         & = (\mu(g), \lambda(g)) \cdot(\mu(h), \lambda(h))    \\
                         & = (\mu(g) + \lambda(g)\mu(h), \lambda(g)\lambda(h)) \\
        \implies \mu(gh) & = \mu(g) + \lambda(g)\mu(h)
    \end{align*}
    and so $\mu$ is a crossed homomorphism for $\lambda$.
\end{proof}

\begin{definition}[Principal Crossed Homomorphism]
    A \textit{principal crossed homomorphism} for $\lambda$ is a crossed homomorphism
    of the form $\mu(g) = y- \lambda(g)y$ for some $y \in E$.
\end{definition}
\begin{remark}
    If $\mu$ is a principal crossed homomorphism for $\lambda$ determined by some $y\in E$, we will often write $\mu = D_y$. This comes from the fact that
    principal crossed homomorphisms are sometimes also called \emph{principal derivations}.
\end{remark}

\begin{lemma}
    The group cohomology $H^1(G, E)$ is isomorphic to
    \begin{center}
        $H^1(G,E) \cong \dfrac{\{\text{crossed homomorphisms for } \lambda\}}{\{\text{principal crossed homomorphisms for } \lambda\}}$
    \end{center}
    \label{lemma:crossed-coho}
\end{lemma}

\begin{proof}
    This is a standard result in group cohomology. See for example \cite{tomczak2023groupcohomology}.
\end{proof}

\subsection{Radiant affine manifolds}
Of importance to completeness of the affine structure is the case that our
affine representation has a fixed-point. It turns out that such structures will
always be incomplete.
\begin{definition}
    An affine representation with a fixed-point is called \textit{radiant}.
\end{definition}
\begin{prop}
    Let $y\in E$. Then $y$ is a \textit{stationary point} of the action of $\alpha$ if
    and only if $\mu(g) = y - \lambda(g)y$ for all $g \in G$.
    \label{prop:ary-point}\label{lemma:stationary}
\end{prop}

\begin{proof}
    Suppose $y \in E$ is a stationary point of $\alpha$. Then $\alpha(g)y = y$ for every $g \in G$.
    This gives that
    \begin{align*}
        (\lambda(g) + \mu(g))y & = \lambda(g)y + \mu(g) \\
                               & = y                    \\
        \implies \mu(g)        & = y - \lambda(g)y
    \end{align*}

    Conversely, suppose that $\mu(g) = y - \lambda(g)y$. Then $y = \mu(g) +
        \lambda(g)y = \alpha(g)y$
\end{proof}

\begin{definition}[Radiance Obstruction]
    Let $\alpha$ be an affine representation. The radiance obstruction of $\alpha$ is the cohomology
    class
    $$
        c_{\alpha} = [\mu] \in H^1(G, E)
    $$
    where $\mu$ is the translational part of $\alpha$.
\end{definition}

\begin{lemma} \label{lemma:radiance-obstruction}
    The radiance obstruction vanishes, e.g. $c_{\alpha} = 0$ if and only if
    $\alpha$ is conjugate to its linear part by a translation, if and only if $\alpha$
    has a stationary point.
\end{lemma}

\begin{proof}
    By Lemma~\ref{lemma:stationary}, $y \in E$ is stationary if and only if
    $\mu(g) = D_y$ is a principal crossed homomorphism for $\lambda$ if and only
    if $c_{\alpha} = [\mu] =0$.

    Now suppose that $\mu =D_y$ for some $y \in E$ so that $\mu(g) = y -
        \lambda(g)y$. We define the translation
    \begin{center}
        $T_y : E \to E$\\
        $ x \mapsto x-y.$
    \end{center}
    Now, conjugating $\alpha$ by this translation yields
    \begin{align*}
        T_y \circ \alpha(g) \circ T_y^{-1}(x) & = T_y\circ \alpha(g)(x+y)                                \\
                                              & = T_y\circ(\lambda(g)x  + \lambda(g)y + \mu(g))          \\
                                              & = T_y \circ (\lambda(g)x + \lambda(g)y +y - \lambda(g)y) \\
                                              & = T_y \circ (\lambda(g)x +y )                            \\
                                              & = \lambda(g)x +y-y                                       \\
                                              & = \lambda(g)x.
    \end{align*}

    Conversely, suppose that $T_y \circ \alpha(g) \circ T_y^{-1} = \lambda(g)$ for
    all $g \in G$ and $x \in E$. Then
    \begin{align*}
        T_y \circ \alpha(g) \circ T_y(x) & = T_y \circ \alpha(g)(x+y)             \\
                                         & = T_y \circ (\lambda(g)(x+y) + \mu(g)) \\
                                         & = \lambda(g)(x+y)+\mu(g)-y             \\
                                         & = \lambda(g)x
    \end{align*}
    and so
    \begin{align*}
        \mu(g) & = \lambda(g)(x) - \lambda(g)(x) - \lambda(g)(y) +y \\
               & = y - \lambda(g)y
    \end{align*}
    as required.
\end{proof}

    This lemma will allow us to alter the development map by a translation (which
    alters the holonomy representation by a conjugation) to assume that our fixed

\begin{definition}[Contragradient Representation]
    Let $E$ be a $G$-module by $\lambda: G \to \text{Aut}(V)$. We define
    the contragradient representation on $E^*$ by
    \begin{center}
        $\lambda^*: G \to GL(E^*)$\\
        $g \mapsto \lambda^* g$
    \end{center}
    such that $\lambda^*g(f)(x) = f(\lambda(g^{-1})x)$ for all $x \in E$ and $f\in E^*$.
\end{definition}
\begin{remark}
    Once a representation $\lambda: G \to \text{Aut}(E)$ has been fixed, we will often write the action of $g$ on an element $x$ of $E$ or $E^*$ as
    $g \cdot x$. It is clear from if $x$ is an element of $E$ or $E^*$ whether this action should be taken as the action of $\lambda(g)$ or $\lambda^*g$.
\end{remark}
\begin{remark}
    This induces a representation on $\wedge^kE^*$ by \\$(g\cdot \omega)(x_1, \dots , x_n) = \omega(g^{-1}\cdot x_1, \dots, g^{-1}\cdot x_n)$ for $\omega \in \wedge^k E^*$
        and $x_i \in E$.
\end{remark}

We will make use of the following lemma when looking at the relationship
between the holonomy representation and parallel volume forms.

\begin{lemma}
    Let $F: V \to V$ be a linear map from a vector space $V$ of dimension $n$ to itself
    and let $\omega \in \wedge^n(V^*)$. Then
    \begin{center}
        $F^*\omega = \text{det}(A)\omega$
    \end{center}
    where $A$ is any matrix representing the linear transformation $F$.
    \label{lemma:pullback-form-linearmap}
\end{lemma}
\begin{proof}
    As $\text{dim}(\wedge^n(V^*)) = 1$ and $F$ is a linear map, it follows that our map corresponds to multiplication
    by some scalar $\lambda \in \mathbb{R}$. Let $\varphi: V \to \mathbb{R}^n$ be an isomorphism identifying
    our vector space $V$ with $\mathbb{R}$. Then we have
    \begin{align*}
        F^*\varphi^*\det                           & = \lambda\varphi^* \det \\
        \implies (\varphi^{-1})^* F^*\varphi^*\det & = \lambda \det          \\
        \implies (\varphi^{-1} F \varphi)^*\det    & = \lambda \det.
    \end{align*}
    If we let $A = \varphi^{-1} F \varphi$ and $\{e_1, \dots , e_n\}$ the standard basis of $\mathbb{R}^n$
    then
    \begin{align*}
        A^*\det(e_1. \dots , e_n) & =  \det(Ae_1, \dots, Ae_n)       \\
                                  & = \lambda \det(e_1, \dots , e_n) \\
                                  & = \lambda
    \end{align*}
    which gives us that $\det(A)= \lambda$ as required.
\end{proof}

This leads us to the following elementary but important property of the action
of the holonomy representation on volume forms.
\begin{corollary}
    Let $M$ be an affine manifold and $\lambda: G \to  GL(\mathbb{R}^n)$ be the linear part of the holonomy
    representation. Then $M$ has a parallel volume form with respect to the affine connection if
    and only if $\det(\lambda(g)) = 1$ for all $g \in G$.\label{lemma:volume}
\end{corollary}

\begin{proof}
    Suppose $\omega$ is a parallel volume form on $M$. Then the parallel transport of $\omega$ around a loop in $M$
    leaves $\omega$ unchanged, e.g. the linear holonomy action preserves $\omega$. This gives us that
    \begin{center}
        $g\cdot \omega = \omega$\\
        $\implies \omega(g^{-1}\cdot x_1, \dots, g^{-1}\cdot x_n) = \omega(x_1, \dots, x_n)$\\
    \end{center}
    for all $x_i \in M$. But by Lemma \ref{lemma:pullback-form-linearmap}
    \begin{center}
        $\omega(g^{-1}\cdot x_1, \dots, g^{-1}\cdot x_n) = \det(\lambda(g^{-1}))\cdot \omega(x_1, \dots, x_n)$
    \end{center}
    and so the volume is preserved if $\det(\lambda(g)) = 1$

    Conversely, suppose that the determinant of the linear holonomy is $1$ for all
    $g \in G$. If we pick a point $m_0 \in M$ then locally there exists a volume
    form $\omega_0$ on a chart $U$ of $M$ with respect to the affine connection,
    corresponding to a constant, top degree form. We now claim that we can uniquely
    extend such a form. Indeed, as parallel transport provides a linear isomorphism
    between tangent spaces we can attempt to define a volume form on $M$ by
    parallel transporting $\omega_0$.

    Pick another point $m \in M$. We claim that the parallel transport of
    $\omega_0$ along a curve $\gamma$ from $m_0$ to $m$ does not depend on the
    choice of curve. If $\gamma'$ is another curve then the parallel transport
    $(P^{\gamma'}_{m_0 \to m})^{-1} \circ P^{\gamma}_{m_0 \to m} = g$ for some $g$
    in the linear holonomy group. It follows that
    \begin{align*}
        g^* \circ \omega_0                                 & = ((P^{\gamma'}_{m_0 \to m})^{-1} \circ P^{\gamma}_{m_0 \to m})^* \circ \omega_0 \\
                                                           & = \det(\lambda(g))\cdot \omega_0                                                 \\
                                                           & = \omega_0                                                                       \\
        \implies (P^{\gamma}_{m_0 \to m})^* \circ \omega_0 & = (P^{\gamma'}_{m_0 \to m})^* \circ \omega_0
    \end{align*}
    and so the parallel transport does from $m$ to $m_0$ does not depend on the choice of path. We note
    that this defines a continuous maps as in each affine chart the parallel transport identifies the tangent
    spaces via the identity map and on over laps they differ by an element of $GL(\mathbb{R}^n)$ as the derivative
    of an affine map extracts the linear part.
\end{proof}

\begin{definition}[Radiant Manifold]
    An affine manifold is said to be radiant if its affine holonomy representation has a fixed-point.
\end{definition}

\begin{remark}
    By Lemma \ref{lemma:radiance-obstruction} we may conjugate our developing map by a translation to assume that
    the fixed-point is at the origin.
\end{remark}

When $M$ is radiant the above remark implies that the radiant vector field $R =
    \sum_i x_i \frac{\partial}{\partial x_i}$ is fixed by the affine action. There
is then a unique vector field $\tilde{X}$ on the universal cover $\tilde{M}$
that is related to $R$ by the developing map. Since this is preserved by deck
transformations (by equivariance of developing map and since the representation
is radiant) it descends to a vector field $X$ on $M$.

\begin{definition}
    The vector field $X$ on $M$ is called the radiant vector field of $M$.
\end{definition}

\section{Integral Affine Manifolds}
One of the main objects of interest in this thesis is integral affine
manifolds. These are $(G,X)$-structures whereby $G=\affz(\R^n)$ and $X= \R^n$.
They show up in symplectic geometry as the base space of a regular Lagrangian
torus fibration inherits such a structure as we explain in Chapter
\ref{LagrangianFibrations}. They are also important objects in the study of the
SYZ conjecture in physics which relates a six-dimensional Calabi-Yau manifold
$X$ to its so-called mirror $\tilde{X}$ by a Lagrangian fibration over a common
base space $B$ \cite{SYZ}. In this chapter we give some basic properties of
these manifolds as well as some examples.

\begin{prop}\label{prop:integral-affine-equiv-lattice}
    Let $M$ be a smooth manifold. An integral affine structure on $M$ is equivalent to a full-rank lattice
    $\Lambda \subset T^*M$ that is locally spanned by closed one-forms.
\end{prop}
\begin{proof}
    Given an integral affine structure on $M$, we can define a lattice in $T^*M$ as follows. Let $(U, x_1, \dots ,x_n)$ be a chart on $M$. We define
    $\Lambda|_U = \text{span}_{\Z}(dx_1, \dots, dx_n)$. Carrying out this construction in each affine chart defines a full-rank lattice in $T^*M$. On overlaps, these lattices
    will differ by the differential of an integral affine map $x \mapsto Ax + b$ with $A \in \text{Gl}(\Z^n)$ and $b \in \R^n$. But the differential of such a map is given by the linear part $A$ and hence is an automorphism of the lattice.

    Conversely, given a full-rank lattice $\Lambda \subset T^*M$ that is locally
    spanned by closed one-forms, by shrinking charts if necessary we can assume
    that our lattice is spanned by exact one-forms $(dx_1, \dots , dx_n)$ in an
    open neighborhood $U$. Since the $dx_i|_p$ span a full-rank lattice in
    $T_p^*M$, they are linearly independent and hence if we define coordinates by
    $(U, x_1 ,\dots , x_n)$ by the inverse function theorem, the map $\bar{x} =
        (x_1, \dots ,x_n)$ is a local diffeomorphism. On the overlap of two charts $U_i
        \cap U_j$, we know that the pullback of the lattice in $U_i$ to $U_j$ must be
    an automorphism of the lattice. But these maps are given exactly by elements of
    the integral affine group $\affz(\R^n)$.
\end{proof}

We now give some examples of integral affine manifolds.
\begin{eg}
    Let $\Z^n$ act on $\R^n$ by translations. Then the quotient $(\R/\Z)^n$ inherits an integral affine structure and is the standard integral affine torus. We note that
    this integral affine structure cannot be equivalent to any of the structures obtained from Example \ref{eg:hopf-manifold} as these are incomplete. In fact, it is also true that
    a closed integral affine manifold cannot be radiant \cite{nilpotent-holonomy}.
\end{eg}

\begin{eg}
    Let $B$ be an integral affine manifold and let $f: M \to B$ be a local diffeomorphism where $M$ is any smooth manifold. Then there is a unique integral affine structure on $M$ such that $f$ is
    an integral affine map as defined in Definition \ref{def:locally-g}. The developing map is an example of such a map and in this sense the integral affine structure on any integral affine manifold
    can be realised as the pullback of the standard integral affine structure on $\R^n$.
\end{eg}

\begin{eg}
    Let $M$ be an integral affine manifold and let $\Gamma$ be a discrete group acting freely and properly on $M$ by integral affine diffeomorphisms. Then the quotient
    $M/\Gamma$ inherits an integral affine structure and the projection map $\pi: M \to M/\Gamma$ is an integral affine map.
\end{eg}

We can of course define an integral affine structure by specifying a full-rank
lattice in $T^*M$ that is locally spanned by closed one-forms. Importantly, the
cotangent bundle of any manifold $M$ inherits a symplectic structure which
gives rise to a volume form $\omega$ on $T^*M$. If $\Lambda, \Lambda' \subset
    T^*M$ are two lattices in $T^*M$ such that they give $M$ an integral affine
structure, then the volume form descends to the quotient and the induced volume
of $T^*M/\Lambda$ and $T^*M/\Lambda'$ must be the same if $\Lambda$ and
$\Lambda'$ determine equivalent integral affine structures. This will be
discussed more in Chapter \ref{LagrangianFibrations}.

\begin{eg}[Radiant integral affine manifold]
    If we drop the compactness assumption, it is possible to equip a manifold $M$ with an integral affine structure that is radiant.
    Let $M= \R^2 - \{0\}$ and
    $$
        i :  M \hookrightarrow \R^2
    $$
    be the inclusion. Then this is a local diffeomorphism and hence, equipping $\R^2$ with the standard integral affine structure, this induces an integral affine structure on $M$.
    There is a covering map
    $$
        \pi: (0, \infty) \times \R = \tilde{M} \to M,
    $$
    $$ (r, \theta) \mapsto (r \cos \theta, r \sin \theta)$$
    and the composition $\widetilde{\dev} = i \circ \pi : \tilde{M}  \to \R^2$ is the developing map for the integral affine structure induced on $M$. This developing map fails to be injective and surjective,
    hence this affine structure is not complete and is therefore not affinely diffeomorphic to the standard integral affine structure on $\R^2$. By considering $\theta \mapsto \theta + 2\pi$ one
    notes that $\widetilde{\dev}(r, \theta + 2\pi) = \text{id}_{\R^2} \widetilde{\dev}(r, \theta)$
    and so the holonomy representation is the trivial one in this case.

    We can modify the affine structure so that it becomes radiant as follows. For
    any $A\in \text{GL}(\Z^2)$, there is a free and proper action of $\Z$ on
    $\tilde{M}$ by integral affine diffeomorphisms which acts by covering the
    multiplication by $A^n$ on $M$ and translating a sheet as shown in
    \cite{Bates2007}. It follows that the quotient of $\tilde{M}$ by this action is
    an integral affine manifold. Its holonomy representation satisfies
    \begin{align*}
        \hol (\gamma) = (A, 0)
    \end{align*}
    where $\gamma$ is the generator of $\pi_1(M)$. It follows that this integral affine structure is radiant.
\end{eg}

\begin{eg}[Mapping tori]
    Let $M$ be an integral affine manifold and let $f: M \to M$ be an integral affine diffeomorphism. Then, equipping $\R$ with its
    standard integral affine structure, we can form the quotient manifold
    $$
        M_f = \dfrac{M \times \R}{\sim}
    $$
    where $(m,t) \sim (m', t')$ if $(m,t) = (f^{-n}(m'), t' + n)$ for some $n \in \Z$ where $f^n$ denotes composition of $f$ or $f^{-1}$ depending on if $n$ is positive. We note that even if $f$ is not free,
    the $\Z$-action defining the quotient is free as it is a translation in the second component. The resulting manifold is called the \emph{mapping torus} of $f$. We note that this is a fiber bundle over $S^1$ through projection
    onto the second component.
\end{eg}
\begin{remark}
    If $\Sigma$ is compact and admits a $\R P^n$-structure and $f$ is a projective automorphism, then the mapping torus of $f$ admits a radiant affine structure \cite{goldman2022geometric}.
\end{remark}
